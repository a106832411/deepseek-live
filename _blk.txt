   private static int CountCrops(GameLocation location)
        {
            if (location == null) return 0;
            int n = 0;
            foreach (var kv in location.terrainFeatures.Pairs)
            {
                if (kv.Value is StardewValley.TerrainFeatures.HoeDirt d && d.crop != null) n++;
            }
            return n;
        }
