def walk_until(tx, ty, timeout=10):
    send({"actionType": "player_walk_to", "x": tx, "y": ty}, 0.25)
    t0 = time.time()
    while time.time() - t0 < timeout:
        time.sleep(0.3)
        b = rd()
        if b and tile(b) == (tx, ty):
            return True
    return False
